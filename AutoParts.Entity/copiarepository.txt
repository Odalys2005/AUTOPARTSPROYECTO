    public class CategoriaRepository
    {
        private readonly ConnectionFactory _connectionFactory;

        public CategoriaRepository(ConnectionFactory connectionFactory)
        {
            _connectionFactory = connectionFactory;
        }

        public List<Categoria> SelectAll()
        {
            List<Categoria> result = new List<Categoria>();

            using (SqlConnection conn =
                _connectionFactory.CreateConnection())
            {
                using (SqlCommand cmd =
                    new SqlCommand("spCategoriaSelectAll", conn))
                {
                    cmd.CommandType = CommandType.StoredProcedure;

                    conn.Open();

                    using (SqlDataReader dr =
                        cmd.ExecuteReader(CommandBehavior.SingleResult))
                    {
                        while (dr.Read())
                        {
                            Categoria entity = new Categoria();

                            entity.CategoriaId = dr.GetInt32(0);
                            entity.Nombre = dr.GetString(1);
                            entity.EstadoId = dr.GetInt32(2);

                            result.Add(entity);
                        }
                    }
                }
            }

            return result;
        }

        public int Insert(Categoria entity)
        {
            using (SqlConnection conn =
                _connectionFactory.CreateConnection())
            {
                using (SqlCommand cmd =
                    new SqlCommand("spCategoriaInsert", conn))
                {
                    cmd.CommandType = CommandType.StoredProcedure;

                    cmd.Parameters.Add("@Nombre", SqlDbType.VarChar, 100)
                        .Value = entity.Nombre;

                    cmd.Parameters.Add("@EstadoId", SqlDbType.Int)
                        .Value = entity.EstadoId;

                    conn.Open();
                    cmd.ExecuteNonQuery();

                    return 1;
                }
            }
        }

        public int Update(Categoria entity)
        {
            using (SqlConnection conn =
                _connectionFactory.CreateConnection())
            {
                using (SqlCommand cmd =
                    new SqlCommand("spCategoriaUpdate", conn))
                {
                    cmd.CommandType = CommandType.StoredProcedure;

                    cmd.Parameters.Add("@CategoriaId", SqlDbType.Int)
                        .Value = entity.CategoriaId;

                    cmd.Parameters.Add("@Nombre", SqlDbType.VarChar, 100)
                        .Value = entity.Nombre;

                    cmd.Parameters.Add("@EstadoId", SqlDbType.Int)
                        .Value = entity.EstadoId;

                    conn.Open();

                    cmd.ExecuteNonQuery();

                    return 1;
                }
            }
        }

        public int Delete(Categoria entity)
        {
            using (SqlConnection conn =
                _connectionFactory.CreateConnection())
            {
                using (SqlCommand cmd =
                    new SqlCommand("spCategoriaDelete", conn))
                {
                    cmd.CommandType = CommandType.StoredProcedure;

                    cmd.Parameters.Add("@CategoriaId", SqlDbType.Int)
                        .Value = entity.CategoriaId;

                    conn.Open();

                    using (SqlDataReader dr = cmd.ExecuteReader())
                    {
                        if (!dr.Read())
                            throw new InvalidOperationException(
                                "SQL Server no confirmó la desactivación.");

                        int estadoId = dr.GetInt32(
                            dr.GetOrdinal("EstadoId"));

                        if (estadoId != 2)
                            throw new InvalidOperationException(
                                "La categoría no quedó inactiva.");
                        return estadoId;
                    }
                }
            }
        }
    }

    public class CompraRepository

                    {

        // Se conserva para compatibilidad con otras pantallas existentes.

                using (SqlCommand cmd = new SqlCommand("spCompraInsert", conn))

                        .Value = entity.FechaCompra;

        // Registra cabecera, detalles y aumentos de inventario mediante
        // UNA transaccion controlada en dbo.spCompraRegistrarCompleta.
        // No llamar tambien a Insert(Compra) ni a Detalle_CompraRepository.Insert().
        public CompraRegistroResultado RegistrarCompleta(
            Compra compra,
            IReadOnlyCollection<Detalle_Compra> detalles)
        {
            if (compra == null)
                throw new ArgumentNullException(nameof(compra));
            if (detalles == null)
                throw new ArgumentNullException(nameof(detalles));
            if (compra.ProveedorId <= 0 || compra.EmpleadoId <= 0)
                throw new ArgumentException("Selecciona un proveedor y un empleado válidos.");

            string comprobante = (compra.NumeroComprobante ?? "").Trim();
            if (comprobante.Length == 0 || comprobante.Length > 50)
                throw new ArgumentException("Ingresa un número de comprobante de hasta 50 caracteres.");
            if (compra.FechaCompra == default ||
                compra.FechaCompra < new DateTime(1753, 1, 1) ||
                compra.FechaCompra > new DateTime(9999, 12, 31))
                throw new ArgumentException("Ingresa una fecha de compra válida para SQL Server.");
            if (detalles.Count == 0)
                throw new ArgumentException("Agrega al menos un producto a la compra.");
            if (detalles.Any(d => d == null || d.ProductoId <= 0 || d.Cantidad <= 0 ||
                                  d.PrecioCosto < 0 || decimal.Round(d.PrecioCosto, 2) != d.PrecioCosto))
                throw new ArgumentException("Revisa los productos, cantidades y precios de costo.");
            if (detalles.GroupBy(d => d.ProductoId).Any(g => g.Count() > 1))
                throw new ArgumentException("Un producto se repite: unifica sus cantidades.");

            // La base de datos recalcula los subtotales y el total; no se
            // confia en importes enviados por el HTML ni por el llamador.
            string detallesJson = JsonSerializer.Serialize(detalles.Select(d => new
            {
                d.ProductoId,
                d.Cantidad,
                d.PrecioCosto
            }).ToArray());

            using (SqlConnection conn = _connectionFactory.CreateConnection())
            using (SqlCommand cmd = new SqlCommand("dbo.spCompraRegistrarCompleta", conn))
            {
                cmd.CommandType = CommandType.StoredProcedure;
                cmd.CommandTimeout = 60;
                cmd.Parameters.Add("@ProveedorId", SqlDbType.Int).Value = compra.ProveedorId;
                cmd.Parameters.Add("@EmpleadoId", SqlDbType.Int).Value = compra.EmpleadoId;
                cmd.Parameters.Add("@NumeroComprobante", SqlDbType.VarChar, 50).Value = comprobante;
                cmd.Parameters.Add("@FechaCompra", SqlDbType.DateTime).Value = compra.FechaCompra;
                cmd.Parameters.Add("@Detalles", SqlDbType.NVarChar, -1).Value = detallesJson;

                conn.Open();
                using (SqlDataReader dr = cmd.ExecuteReader(CommandBehavior.SingleRow))
                {
                    if (!dr.Read())
                        throw new InvalidOperationException(
                            "SQL Server no devolvió la confirmación de la compra. Verifica el estado antes de reintentar.");

                    return new CompraRegistroResultado
                    {
                        CompraId = dr.GetInt32(dr.GetOrdinal("CompraId")),
                        NumeroComprobante = dr.GetString(dr.GetOrdinal("NumeroComprobante")),
                        TotalCompra = dr.GetDecimal(dr.GetOrdinal("TotalCompra")),
                        AdvertenciasStockMaximo = dr.IsDBNull(dr.GetOrdinal("AdvertenciasStockMaximo"))
                            ? ""
                            : dr.GetString(dr.GetOrdinal("AdvertenciasStockMaximo"))
                    };
                }
            }
        }
    }

    // Resultado de la operacion completa, sin exponer el mecanismo SQL al Dashboard.
    public class CompraRegistroResultado
    {
        public int CompraId { get; set; }
        public string NumeroComprobante { get; set; } = "";
        public decimal TotalCompra { get; set; }
        public string AdvertenciasStockMaximo { get; set; } = "";
    }

            int cantidadReclamada,
            string descripcionFalla,
            string? observaciones = null)
        {
            using (SqlConnection conn = _connectionFactory.CreateConnection())
            using (SqlCommand cmd = new SqlCommand("dbo.spGarantiaRegistrarSolicitud", conn))

                            ReemplazoGarantia entity = new ReemplazoGarantia();

                            entity.GarantiaId = dr.GetInt32(1);
                            entity.EmpleadoId = dr.GetInt32(2);
                            entity.FechaReemplazo = dr.GetDateTime(3);
                            entity.Cantidad = dr.GetInt32(4);

                            entity.Observaciones = dr.IsDBNull(5)
                                ? null
                                : dr.GetString(5);

                            result.Add(entity);
                        }
                    }
                }
            }

            return result;
        }

        // Registra la entrega real en una transacción SQL y descuenta stock.
        // Si no hay existencias, el procedimiento devuelve PENDIENTE_REEMPLAZO
        // sin generar un registro de entrega ni descontar inventario.
        public ReemplazoEntregaResultado EntregarReemplazo(
            int garantiaId, int empleadoId, string? observaciones = null)
        {
            if (garantiaId <= 0)
                throw new ArgumentException("Selecciona una garantía válida.", nameof(garantiaId));
            if (empleadoId <= 0)
                throw new ArgumentException("Selecciona un empleado válido.", nameof(empleadoId));
            if (observaciones != null && observaciones.Length > 1000)
                throw new ArgumentException("Las observaciones superan 1000 caracteres.", nameof(observaciones));

            using (SqlConnection conn = _connectionFactory.CreateConnection())
            using (SqlCommand cmd = new SqlCommand("dbo.spGarantiaEntregarReemplazo", conn))
            {
                cmd.CommandType = CommandType.StoredProcedure;
                cmd.CommandTimeout = 45;
                cmd.Parameters.Add("@GarantiaId", SqlDbType.Int).Value = garantiaId;
                cmd.Parameters.Add("@EmpleadoId", SqlDbType.Int).Value = empleadoId;
                cmd.Parameters.Add("@Observaciones", SqlDbType.VarChar, 1000).Value =
                    string.IsNullOrWhiteSpace(observaciones)
                        ? DBNull.Value
                        : observaciones.Trim();

                conn.Open();
                using (SqlDataReader dr = cmd.ExecuteReader(CommandBehavior.SingleRow))
                {
                    if (!dr.Read())
                        throw new InvalidOperationException(
                            "SQL Server no devolvió confirmación del reemplazo. Verifica el estado antes de reintentar.");

                    int colId = dr.GetOrdinal("ReemplazoId");
                    return new ReemplazoEntregaResultado
                    {
                        ReemplazoId = dr.IsDBNull(colId) ? null : dr.GetInt32(colId),
                        GarantiaId = dr.GetInt32(dr.GetOrdinal("GarantiaId")),
                        ProductoId = dr.GetInt32(dr.GetOrdinal("ProductoId")),
                        Cantidad = dr.GetInt32(dr.GetOrdinal("Cantidad")),
                        StockRestante = dr.GetInt32(dr.GetOrdinal("StockRestante")),
                        EstadoGarantia = dr.GetString(dr.GetOrdinal("EstadoGarantia")),
                        Mensaje = dr.GetString(dr.GetOrdinal("Mensaje"))
                    };
                }
            }
        }

        // LEGADO: este procedimiento NO descuenta inventario.
        // No usar este método en el flujo integrado de garantías.
        public int Insert(ReemplazoGarantia entity)
        {
            int result = 0;

            using (SqlConnection conn = _connectionFactory.CreateConnection())
            {
                using (SqlCommand cmd = new SqlCommand("spReemplazoGarantiaInsert", conn))
                {
                    cmd.CommandType = CommandType.StoredProcedure;

                    cmd.Parameters.Add("@GarantiaId", SqlDbType.Int)
                        .Value = entity.GarantiaId;

                    cmd.Parameters.Add("@EmpleadoId", SqlDbType.Int)
                        .Value = entity.EmpleadoId;

                    cmd.Parameters.Add("@FechaReemplazo", SqlDbType.DateTime)
                        .Value = entity.FechaReemplazo;

                    cmd.Parameters.Add("@Cantidad", SqlDbType.Int)
                        .Value = entity.Cantidad;

                    cmd.Parameters.Add("@Observaciones", SqlDbType.VarChar, 1000)
                        .Value = (object?)entity.Observaciones ?? DBNull.Value;

                    conn.Open();

                    result = cmd.ExecuteNonQuery();
                }
            }

            return result;
        }
    }

                    cmd.Parameters.Add("@Anio", SqlDbType.SmallInt)
                        .Value = entity.Anio;

                    SqlParameter cilindrada = cmd.Parameters.Add(
                        "@Cilindrada", SqlDbType.Decimal);

                    cilindrada.Precision = 5;
                    cilindrada.Scale = 2;
                    cilindrada.Value = entity.Cilindrada;

                    cmd.Parameters.Add("@TipoMotor", SqlDbType.VarChar, 100)
                        .Value = entity.TipoMotor;

                    cmd.Parameters.Add("@EstadoId", SqlDbType.Int)
                        .Value = entity.EstadoId;

                    conn.Open();

                    result = cmd.ExecuteNonQuery();
                }
            }

            return result;
        }

        public int Delete(Vehiculo entity)
        {
            using (SqlConnection conn =
                _connectionFactory.CreateConnection())
            {
                using (SqlCommand cmd =
                    new SqlCommand("spVehiculoDelete", conn))
                {
                    cmd.CommandType = CommandType.StoredProcedure;

                    cmd.Parameters.Add("@VehiculoId", SqlDbType.Int)
                        .Value = entity.VehiculoId;

                    conn.Open();

                    using (SqlDataReader dr = cmd.ExecuteReader())
                    {
                        if (!dr.Read())
                            throw new InvalidOperationException(
                                "SQL Server no confirmó la desactivación.");

                        int estadoId = dr.GetInt32(
                            dr.GetOrdinal("EstadoId"));

                        if (estadoId != 2)
                            throw new InvalidOperationException(
                                "El Vehiculo no quedó inactivo.");
                        return estadoId;
                    }
                }
            }
        }

    public class VentaRepository
    {
        private readonly ConnectionFactory _connectionFactory;

        public VentaRepository(ConnectionFactory connectionFactory)
        {
            _connectionFactory = connectionFactory;
        }

        public List<Venta> SelectAll()
        {
            List<Venta> result = new List<Venta>();

            using (SqlConnection conn = _connectionFactory.CreateConnection())
            {
                using (SqlCommand cmd = new SqlCommand("spVentaSelectAll", conn))
                {
                    cmd.CommandType = CommandType.StoredProcedure;

                    conn.Open();

                    using (SqlDataReader dr = cmd.ExecuteReader(CommandBehavior.SingleResult))
                    {
                        while (dr.Read())
                        {
                            Venta entity = new Venta();

                            entity.VentaId = dr.GetInt32(0);
                            entity.ClienteId = dr.GetInt32(1);
                            entity.EmpleadoId = dr.GetInt32(2);
                            entity.NumeroFactura = dr.GetString(3);
                            entity.FechaVenta = dr.GetDateTime(4);
                            entity.Total = dr.GetDecimal(5);
                            entity.EstadoId = dr.GetInt32(6);

                            result.Add(entity);
                        }
                    }
                }
            }

            return result;
        }

        // Registra cabecera + detalles + salida de inventario en UNA transacción SQL.
        public VentaRegistroResultado RegistrarCompleta(VentaRegistroCompleto solicitud)
        {
            if (solicitud == null) throw new ArgumentNullException(nameof(solicitud));

            DataTable lineas = new DataTable();
            lineas.Columns.Add("ProductoId", typeof(int));
            lineas.Columns.Add("Cantidad", typeof(int));
            foreach (VentaLineaRegistro linea in solicitud.Lineas)
                lineas.Rows.Add(linea.ProductoId, linea.Cantidad);

            using (SqlConnection conn = _connectionFactory.CreateConnection())
            using (SqlCommand cmd = new SqlCommand("dbo.spVentaRegistrarCompleta", conn))
            {
                cmd.CommandType = CommandType.StoredProcedure;
                cmd.CommandTimeout = 45;
                cmd.Parameters.Add("@ClienteId", SqlDbType.Int).Value = solicitud.ClienteId;
                cmd.Parameters.Add("@EmpleadoId", SqlDbType.Int).Value = solicitud.EmpleadoId;
                cmd.Parameters.Add("@NumeroFactura", SqlDbType.VarChar, 50).Value =
                    string.IsNullOrWhiteSpace(solicitud.NumeroFactura)
                        ? DBNull.Value
                        : solicitud.NumeroFactura.Trim();
                cmd.Parameters.Add("@EstadoId", SqlDbType.Int).Value = solicitud.EstadoId;

                SqlParameter carrito = cmd.Parameters.Add("@Lineas", SqlDbType.Structured);
                carrito.TypeName = "dbo.AutoPartsVentaLinea";
                carrito.Value = lineas;

                conn.Open();
                using (SqlDataReader dr = cmd.ExecuteReader(CommandBehavior.SingleResult))
                {
                    if (!dr.Read())
                        throw new InvalidOperationException("SQL Server no devolvió la confirmación de la venta.");

                    return new VentaRegistroResultado
                    {
                        VentaId = dr.GetInt32(dr.GetOrdinal("VentaId")),
                        Total = dr.GetDecimal(dr.GetOrdinal("Total")),
                        NumeroFactura = dr.GetString(dr.GetOrdinal("NumeroFactura"))
                    };
                }
            }
        }

        public int Insert(Venta entity)
        {
            int result = 0;

            using (SqlConnection conn = _connectionFactory.CreateConnection())
            {
                using (SqlCommand cmd = new SqlCommand("spVentaInsert", conn))
                {
                    cmd.CommandType = CommandType.StoredProcedure;

                    cmd.Parameters.Add("@ClienteId", SqlDbType.Int)
                        .Value = entity.ClienteId;

                    cmd.Parameters.Add("@EmpleadoId", SqlDbType.Int)
                        .Value = entity.EmpleadoId;

                    cmd.Parameters.Add("@NumeroFactura", SqlDbType.VarChar, 50)
                        .Value = entity.NumeroFactura;

                    cmd.Parameters.Add("@FechaVenta", SqlDbType.DateTime)
                        .Value = entity.FechaVenta;

                    SqlParameter total = cmd.Parameters.Add(
                        "@Total", SqlDbType.Decimal);

                    total.Precision = 18;
                    total.Scale = 2;
                    total.Value = entity.Total;

                    cmd.Parameters.Add("@EstadoId", SqlDbType.Int)
                        .Value = entity.EstadoId;

                    conn.Open();

                    result = cmd.ExecuteNonQuery();
                }
            }

            return result;
        }
